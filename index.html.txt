    <script>
        function openTab(tabId, btn) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
            document.querySelectorAll('nav button').forEach(el => el.classList.remove('active'));
            document.getElementById(tabId).classList.add('active');
            btn.classList.add('active');
        }

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function playTone(freq) {
            if (audioCtx.state === 'suspended') audioCtx.resume();
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'sine';
            osc.frequency.value = freq;
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 1);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + 1);
        }

        const baseFreq = 261.63;
        const notes = ["도", "도#", "레", "레#", "미", "파", "파#", "솔", "솔#", "라", "라#", "시", "도(높음)"];
        const pythRatios = [1, 256/243, 9/8, 32/27, 81/64, 4/3, 729/512, 3/2, 128/81, 27/16, 16/9, 243/128, 2];
        
        const grid = document.getElementById('synthGrid');
        notes.forEach((name, i) => {
            const tetFreq = baseFreq * Math.pow(2, i/12);
            const pythFreq = baseFreq * pythRatios[i];
            
            const card = document.createElement('div');
            card.className = 'note-card';
            card.innerHTML = `
                <div class="note-name">${name}</div>
                <div class="freq-info">12평균율: ${tetFreq.toFixed(1)}Hz<br>피타고라스: ${pythFreq.toFixed(1)}Hz</div>
                <button class="play-btn btn-tet" onclick="playTone(${tetFreq})">12평균율</button>
                <button class="play-btn btn-pyth" onclick="playTone(${pythFreq})">피타고라스</button>
            `;
            grid.appendChild(card);
        });
    </script>
